cmake_minimum_required(VERSION 3.26)
project(lobs_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python 3.11 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND "${Python_EXECUTABLE}" -m nanobind --cmake_dir OUTPUT_VARIABLE NB_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
list(APPEND CMAKE_PREFIX_PATH "${NB_DIR}")
find_package(nanobind CONFIG REQUIRED)
# MuJoCo headers only (struct layouts); the library itself is the one the Python package already loaded.
set(MUJOCO_INCLUDE "" CACHE PATH "directory containing mujoco/mujoco.h")
nanobind_add_module(_native NB_STATIC native.cpp)
target_include_directories(_native PRIVATE "${MUJOCO_INCLUDE}")
if(MSVC)
  target_compile_options(_native PRIVATE /O2 /fp:precise /EHsc)
else()
  target_compile_options(_native PRIVATE -O3 -ffp-contract=off)
endif()
